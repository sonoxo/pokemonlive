// Historical source only: the one-action assertion and four-clip budget are obsolete.
// Explicitly bounded paid QA: sleep -> next-turn sleep -> one KO, plus one
// reusable entrance. Reruns resume saved session IDs; never silently regenerate.
import assert from "node:assert/strict";
import { mkdir, readFile, writeFile } from "node:fs/promises";
import { setTimeout as delay } from "node:timers/promises";
import { createBattle, resolveTurn } from "../src/battle-engine.js";
import { buildAttackRecords, createRealtimeStoryboard } from "../src/attack-storyboard.js";
import { visualScene } from "../src/visual-battle-state.js";

if (!process.argv.includes("--paid")) throw new Error("Use --paid: generates at most four five-second clips on first run.");
const directory = new URL("../.local/state-continuity-qa/", import.meta.url);
await mkdir(directory, { recursive: true });
const statePath = new URL("verification.json", directory);
let saved;
try { saved = JSON.parse(await readFile(statePath, "utf8")); } catch { saved = {}; }
const save = () => writeFile(statePath, JSON.stringify(saved, null, 2));
const base = "http://127.0.0.1:4173";
async function request(path, input) {
  const response = await fetch(base + path, { method: input ? "POST" : "GET", headers: input ? { "Content-Type": "application/json" } : {}, body: input ? JSON.stringify(input) : undefined, signal: AbortSignal.timeout(45000) });
  const payload = await response.json();
  assert(response.ok, payload.error);
  return payload;
}
async function attack(name, result, continuationSessionId = null, commandBridge = false) {
  const attacks = buildAttackRecords(result.events, { battleEpoch: 900, turn: result.state.turn - 1 });
  assert.equal(attacks.length, 1);
  const plan = createRealtimeStoryboard(attacks);
  if (saved[name]?.session?.status === "ready") return saved[name].id;
  if (!saved[name]) {
    const { session } = await request("/api/attack-videos", { version: 2, battleEpoch: 900, turn: attacks[0].turn, attacks, sequence: plan.turn.sequence, presentation: "inline-command-v1", commandBridge, continuationSessionId });
    saved[name] = { id: session.id, attacks };
    await save();
  }
  for (let i = 0; i < 720; i++) {
    const { session } = await request(`/api/attack-videos/${saved[name].id}`);
    if (session.status === "ready") {
      saved[name].session = session;
      await save();
      console.log(JSON.stringify({ case: name, id: session.id, status: session.status, model: session.clips[0].model, generationMs: session.clips[0].generationMs }));
      await delay(1800); // allow diagnostic local archive and final-frame extraction
      return session.id;
    }
    assert.equal(session.status, "generating", JSON.stringify(session));
    await delay(1000);
  }
  throw new Error("QA deadline exceeded; rerun to resume, do not submit again.");
}

let battle = createBattle();
battle.player.active = 2;
battle.opponent.active = 1;
let result = resolveTurn(battle, { type: "move", moveIndex: 2 }, () => .5, { type: "move", moveIndex: 0 });
assert.equal(result.state.opponent.team[1].status, "sleep");
const previous = await attack("sleep", result);
result = resolveTurn(result.state, { type: "move", moveIndex: 1 }, () => .5, { type: "move", moveIndex: 0 });
assert.equal(result.state.opponent.team[1].status, "sleep");
await attack(process.argv.includes("--revision") ? "sleep-next-tail-v2" : "sleep-next-turn", result, previous);
battle = createBattle();
battle.opponent.team[0].hp = 20;
result = resolveTurn(battle, { type: "move", moveIndex: 0 }, () => .5, { type: "move", moveIndex: 0 });
assert.equal(buildAttackRecords(result.events)[0].outcome.fainted, true);
await attack("spiral-ko", result, null, true);

battle = createBattle();
const before = visualScene(battle);
battle.player.active = 1;
const scene = visualScene(battle);
const { job } = await request("/api/scene-videos", { kind: "switch", before, scene });
saved.switch = job;
await save();
for (let i = 0; i < 720; i++) {
  const { job: current } = await request(`/api/scene-videos/${job.key}`);
  if (current.status === "ready") {
    saved.switch = current;
    await save();
    console.log(JSON.stringify({ case: "switch", key: current.key, generationMs: current.generationMs }));
    break;
  }
  assert(!["error", "cancelled"].includes(current.status), current.error);
  await delay(1000);
}
assert.equal(saved.switch.status, "ready", "Switch QA deadline exceeded; rerun to resume, do not submit again.");
